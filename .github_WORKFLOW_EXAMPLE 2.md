Example GitHub Actions workflow (paste into .github/workflows/deploy.yml)

name: CI / Deploy Angular to Firebase

on:
  push:
    branches: [ main ]
  workflow_dispatch:
  schedule:
    - cron: '0 */6 * * *' # optional: every 6 hours to trigger sync

jobs:
  build-and-deploy:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 18
      - name: Install
        run: |
          cd angular && npm ci
      - name: Build
        run: |
          cd angular && npm run build --if-present
      - name: Deploy to Firebase
        uses: w9jds/firebase-action@v2
        with:
          args: deploy --only hosting
        env:
          FIREBASE_TOKEN: ${{ secrets.FIREBASE_TOKEN }}
